  <script>

    async function fetchData() {

      const response = await fetch('https://docs.google.com/spreadsheets/d/e/2PACX-1vS6s54P2VLmnr5XVZtDQD0zhUxNAOEiVwFzdMo_ytqILfmrVO0KfxwVkaLHMNkfK49SBxWhF2d4ivMe/pub?output=csv');

      const data = await response.text();

      const rows = data.split("\n").slice(1);


      let totalRuns = 0;

      let totalWickets = 0;

      rows.forEach(row => {

        const cols = row.split(",");

        totalRuns += parseInt(cols[4]) || 0;

        if (cols[6].trim() === "Yes") totalWickets++;

      });


      const totalBalls = rows.length;

      const overs = Math.floor(totalBalls / 6) + "." + (totalBalls % 6);


      document.getElementById("liveScore").innerText = `Score: ${totalRuns}/${totalWickets} in ${overs} overs`;

    }


    setInterval(fetchData, 5000); // Refresh every 5 seconds

    fetchData();

  </script>
